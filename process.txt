import os
import pandas as pd
import sys
from data_loader import DataLoader
from data_analysis_service import DataAnalysisService

def main():
    current_directory = os.getcwd()
    data_folder_name = 'Olist_Training' 
    path_to_data = os.path.join(current_directory, data_folder_name)

    loader = DataLoader(path_to_data)

    loaded_data = loader.load_csv_files()
    
    if not loaded_data:
        print("Encerrando: Não há dados para processar.")
        sys.exit()

    processor = DataAnalysisService(loaded_data)
    
    processor.print_summaries()
    
    tabela_alvo = 'olist_orders_dataset'
    processor.find_relationships(tabela_alvo)

    processor.analyze_freight_ratio()

if __name__ == "__main__":
    main()
